In diesem Unterkapitel wird ein Plot erstellt.
Ziel dieses Plots ist es die Funktion $sqrt{x}$ zu untersuchen und deren Eigenschaften zu ermitteln.

\begin{figure}[H]
	\centering
	\begin{tikzpicture}%, trim axis right]
		\begin{axis}[
			width=0.7\linewidth, % Die globale Einstellung mit 0.8\linewidth wird einfach überschrieben
			xlabel={$x$},
			ylabel={$f(x)$},
			legend pos=south east, % Dort platzieren, wo keine Messdaten sind!
		]
		% Die Messdaten
			\addplot+[
				only marks,
				mark=*,          % Ausgefüllte Kreise als Markierung
				mark size=1.5pt,  % Etwas kleinere Punkte sehen oft sauberer aus
				y-fehler,
			] table [col sep=semicolon, x=x, y=y, y error=y_err] {test.csv};
			% Der Fit
			\addplot+ [
				mark=none,
				line width=1.2pt,  % Dickere Linie
				domain=0:20,    % x-Bereich
				samples=100,       % Nur 2 Punkte für straight line (super dünn und effizient)
			] { sqrt(x) };	
			% Legende
			\legend{Messwerte, Fit-Funktion ($y = \sqrt{x}$)}
		\end{axis}
	\end{tikzpicture}
	\label{fig:beispielplot}
	\caption{
		Plot der Funktion $f(x) = \sqrt{x}$.
		Erstellt über das TikZ und pgfplot Paket.
	}
